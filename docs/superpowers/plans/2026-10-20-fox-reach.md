**Architecture:** The server (`jotihunt-tracker-server`) computes everything in a new `src/services/reach/` module and serves `GET /routes/reach`; the router gets a `foot_night` profile. The client (`jotihunt-tracker-client`) fetches it with SWR and draws fill/line layers plus HTML marker chips.

## Global Constraints

- Speed: median of usable legs in the last 6 h, clamped 3–7 km/h; fewer than 2 legs: 5 km/h, `measured: false`.
- A leg is skipped when shorter than 10 min, red for more than 30 min during it, a trails `car`/`extra` leg, or (without a trail for it) faster than 8 km/h. Distance: trail `distanceM`, else straight × 1.2.
- Walking time capped at 3 h (`capped: true`).
- Isochrone: `GET /isochrone?point=lat,lng&profile=foot|foot_night&distance_limit=<m>&buckets=1`; `foot_night` when `isNightAt(now, fix)`.
- Router failure: turf circle with the same radius, `estimate: true`.
- Cone: ±30° toward the nearest unvisited group of the same area (straight distance), clipped to the area.
- Areas with status `blue` (OV) get no reach.
- Client layer "Bereik vossen", default off, look A: area fill 0.12, cone fill 0.22, dashed 2 px outline, colour `getColorFromArea`.
- Chip text: `Alpha · 52 min · 4,6 km/u`; `~5 km/u` when not measured; `3u+` when capped; `1u05m` from 60 min.
- Every UI state change animates (fade/scale), including hiding the layer.

```python
def config_hash():
    """sha1 of the shipped config and its custom models: a change means the graph must be imported again."""
    digest = hashlib.sha1()
    with open(CONFIG, "rb") as f:
        digest.update(f.read())
    if os.path.isdir(CUSTOM_MODELS):
        for name in sorted(os.listdir(CUSTOM_MODELS)):
            digest.update(name.encode())
            with open(os.path.join(CUSTOM_MODELS, name), "rb") as f:
                digest.update(f.read())
    return digest.hexdigest()


def served_config():
    """The config the current graph was imported with: its copy, else the legacy config (graphs from before the copy existed)."""
    built = path(BUILT_CONFIG)
    if os.path.isfile(built):

    return LEGACY_CONFIG if os.path.isfile(LEGACY_CONFIG) else CONFIG
```

```ts
    /** The area reachable within `distanceM` along the profile's ways (one bucket). */
    async isochrone({ profile, point, distanceM }) {
      if (!baseUrl) throw new RoutingUnavailableError("ROUTER_URL is not set", { kind: "permanent" });
      const query = new URLSearchParams({ point: `${point[1]},${point[0]}`, profile, distance_limit: String(Math.round(distanceM)), buckets: "1" });
      let res: { status: number; json: unknown };
      try {
        res = await get(`${baseUrl.replace(/\/$/, "")}/isochrone?${query}`, timeoutMs);
      } catch (error) {
        throw new RoutingUnavailableError(`Router unreachable: ${error instanceof Error ? error.message : String(error)}`, { kind: "temporary" });
      }
      const json = (res.json ?? {}) as GhIsochrone;
      if (isRequestError(res.status)) throw new RoutingUnavailableError(json.message ?? `Router answered ${res.status}`, { kind: "request" });
      const geometry = json.polygons?.[0]?.geometry;
      if (res.status !== 200 || !geometry || (geometry.type !== "Polygon" && geometry.type !== "MultiPolygon")) {
        throw new RoutingUnavailableError(json.message ?? `Router answered ${res.status} without a polygon`, { kind: "temporary" });
      }
      return geometry as GeoJSON.Polygon | GeoJSON.MultiPolygon;
```

```ts
import distance from "@turf/distance";
import { statusIntervals, type StatusEvent } from "../routes/trail-legs";

/** Only legs ending in this window count */
export const SPEED_WINDOW_MS = 6 * 3600_000;
/** Shorter legs are GPS/hint jitter */
export const MIN_LEG_MS = 10 * 60_000;
/** A leg with more red than this is a fox that stood still or moved freely, not walking */
export const MAX_RED_MS = 30 * 60_000;
export const MIN_KMH = 3;
export const MAX_KMH = 7;
export const DEFAULT_KMH = 5;
/** Straight distance → walked distance when there is no route */
export const STRAIGHT_FACTOR = 1.2;
/** Without a trail, faster than this was a car */
export const CAR_KMH = 8;
export const MIN_LEGS = 2;

export interface SpeedLeg { id: string; from: Date; to: Date; fromCoord: [number, number]; toCoord: [number, number] }
export interface TrailInfo { mode: "walk" | "car" | "extra"; distanceM: number | null }
export interface Speed { kmh: number; measured: boolean; legs: number }

function redMs(red: { from: Date; to: Date }[], from: Date, to: Date): number {
  let total = 0;
  for (const interval of red) total += Math.max(0, Math.min(interval.to.getTime(), to.getTime()) - Math.max(interval.from.getTime(), from.getTime()));
  return total;
}

function median(values: number[]): number {
  const sorted = [...values].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  return sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
}

/**
 * The fox's walking speed: median km/h of its recent legs, leaving out short legs, legs with a long red period and car
 * moves (trails mode, or faster than CAR_KMH without a trail). Clamped to MIN_KMH-MAX_KMH; DEFAULT_KMH with too few legs.
 * @param trails routed trail info by leg id, or null when trails are off
 */
export function measureSpeed(legs: SpeedLeg[], trails: Map<string, TrailInfo> | null, events: StatusEvent[], now: Date): Speed {
  const red = statusIntervals(events, now).filter((interval) => interval.status === "red");
  const speeds: number[] = [];
  for (const leg of legs) {
    const elapsedMs = leg.to.getTime() - leg.from.getTime();
    if (leg.to.getTime() < now.getTime() - SPEED_WINDOW_MS || elapsedMs < MIN_LEG_MS) continue;
    if (redMs(red, leg.from, leg.to) > MAX_RED_MS) continue;
    const straightM = distance(leg.fromCoord, leg.toCoord, { units: "meters" }) * STRAIGHT_FACTOR;
    const trail = trails?.get(leg.id);
    if (trail && trail.mode !== "walk") continue;
    const kmh = ((trail?.distanceM ?? straightM) / 1000) / (elapsedMs / 3600_000);
    if (!trail && kmh > CAR_KMH) continue;
    speeds.push(kmh);
  }
  if (speeds.length < MIN_LEGS) return { kmh: DEFAULT_KMH, measured: false, legs: speeds.length };
  const kmh = Math.min(MAX_KMH, Math.max(MIN_KMH, median(speeds)));
  return { kmh: Math.round(kmh * 10) / 10, measured: true, legs: speeds.length };
}
```

```ts
import bearing from "@turf/bearing";
import circle from "@turf/circle";
import distance from "@turf/distance";
import { feature, featureCollection } from "@turf/helpers";
import intersect from "@turf/intersect";
import sector from "@turf/sector";
import simplify from "@turf/simplify";

export type AreaGeometry = GeoJSON.Polygon | GeoJSON.MultiPolygon;
export interface Group { teamApiId: number; name: string; coord: [number, number] }
export interface ConeTarget { teamApiId: number; name: string; bearing: number; distanceM: number }

/** Half the cone's opening */
export const CONE_HALF_ANGLE = 30;
/** ~15 m in degrees: isochrones come back with a point every few metres */
const SIMPLIFY_TOLERANCE = 0.00015;

/** Fallback area without a router: a circle of the walking distance. */
export function reachCircle(center: [number, number], distanceM: number): GeoJSON.Polygon {
  return circle(center, distanceM, { units: "meters", steps: 64 }).geometry;
}

export function simplifyArea(area: AreaGeometry): AreaGeometry {
  return simplify(feature(area), { tolerance: SIMPLIFY_TOLERANCE, highQuality: false }).geometry;
}

/** The nearest group (straight distance) not in `visited`. */
export function nearestUnvisited(fix: [number, number], groups: Group[], visited: Set<number>): ConeTarget | null {
  let best: ConeTarget | null = null;
  for (const group of groups) {
    if (visited.has(group.teamApiId)) continue;
    const distanceM = distance(fix, group.coord, { units: "meters" });
    if (!best || distanceM < best.distanceM) best = { teamApiId: group.teamApiId, name: group.name, bearing: bearing(fix, group.coord), distanceM };
  }
  return best && { ...best, distanceM: Math.round(best.distanceM) };
}

/** The ±CONE_HALF_ANGLE wedge from the fix toward the target, cut to the area; null when they do not overlap. */
export function coneIn(fix: [number, number], target: ConeTarget, area: AreaGeometry, radiusM: number): AreaGeometry | null {
  const wedge = sector(fix, radiusM, target.bearing - CONE_HALF_ANGLE, target.bearing + CONE_HALF_ANGLE, { units: "meters", steps: 32 });
  return intersect(featureCollection([wedge, feature(area)]))?.geometry ?? null;
}
```

```ts
import { logger } from "../../logger";
import { Area } from "../../models/area.model";
import { AreaStatusEvent } from "../../models/area-status-event.model";
import { Marker, MarkerType } from "../../models/marker.model";
import { Team } from "../../models/team.model";
import { RoutingUnavailableError } from "../prediction/routing";
import { listCurrentVisits, type VisitDto } from "../prediction/visit.service";
import type { IsochroneRouter, ReachProfile } from "../routes/graphhopper.client";
import { isNightAt } from "../routes/sun";
import type { TrailsResponse } from "../routes/trails.service";
import { coneIn, nearestUnvisited, reachCircle, simplifyArea, type AreaGeometry, type Group } from "./shape";
import { measureSpeed, type SpeedLeg, type TrailInfo } from "./speed";

export interface AreaReach {
  area: string;
  fix: { id: string; type: "hint" | "hunt"; at: string; lng: number; lat: number };
  speedKmh: number; measured: boolean; legs: number;
  distanceM: number; capped: boolean; night: boolean; estimate: boolean;
  polygon: AreaGeometry;
  cone: { teamApiId: number; name: string; bearing: number; distanceM: number; polygon: AreaGeometry } | null;
}
export interface ReachResponse { enabled: true; areas: AreaReach[] }

/** Walking time after which the area stops growing */
export const MAX_WALK_MS = 3 * 3600_000;
/** Distances are rounded to this, so a growing area asks the router about once a minute */
const ROUND_M = 100;
const MEMO_MS = 30_000;
const COOLDOWN_MS = 60_000;
const ISO_CACHE_CAP = 300;

type Coord = [number, number];

export function createReachService(deps: {
  router: IsochroneRouter;
  trails: () => Promise<TrailsResponse>;
  visits?: () => Promise<VisitDto[]>;
  now?: () => Date;
  /** False without ROUTER_URL: circles only */
  configured?: () => boolean;
}) {
  const now = deps.now ?? (() => new Date());
  const configured = deps.configured ?? (() => true);
  const visits = deps.visits ?? listCurrentVisits;
  const isoCache = new Map<string, AreaGeometry>();
  let cooldownUntil = 0;
  let memo: { at: number; value: ReachResponse; body: string } | null = null;
  let inflight: Promise<{ at: number; value: ReachResponse; body: string }> | null = null;

  async function areaShape(fixId: string, profile: ReachProfile, point: Coord, distanceM: number): Promise<{ polygon: AreaGeometry; estimate: boolean }> {
    const key = `${fixId}|${profile}|${distanceM}`;
    const hit = isoCache.get(key);
    if (hit) return { polygon: hit, estimate: false };
    if (configured() && now().getTime() >= cooldownUntil) {
      try {
        const polygon = simplifyArea(await deps.router.isochrone({ profile, point, distanceM }));
        isoCache.set(key, polygon);
        if (isoCache.size > ISO_CACHE_CAP) isoCache.delete(isoCache.keys().next().value!);
        return { polygon, estimate: false };
      } catch (error) {
        if (!(error instanceof RoutingUnavailableError)) logger.error(`Reach isochrone failed: ${error instanceof Error ? error.message : String(error)}`);
        else if (error.kind !== "request") {
          logger.warn(`Router unavailable for reach, circles for ${COOLDOWN_MS / 1000}s: ${error.message}`);
          cooldownUntil = now().getTime() + COOLDOWN_MS;
        }
      }
    }
    return { polygon: reachCircle(point, distanceM), estimate: true };
  }

  async function trailInfo(): Promise<Map<string, TrailInfo> | null> {
    try {
      const { enabled, trails } = await deps.trails();
      if (!enabled) return null;
      return new Map(trails.features.map((f) => [f.properties.id, { mode: f.properties.mode, distanceM: f.properties.distanceM }]));
    } catch (error) {
      logger.warn(`Reach without trails: ${error instanceof Error ? error.message : String(error)}`);
      return null;
    }
  }

  async function compute(): Promise<ReachResponse> {
    const at = now();
    const [areas, markers, events, teams, visitList, trails] = await Promise.all([
      Area.find({}, { name: 1, status: 1 }).lean(),
      Marker.find({ type: { $in: [MarkerType.Hint, MarkerType.Hunt] } }).sort({ time: 1 }).lean(),
      AreaStatusEvent.find().sort({ at: 1 }).lean(),
      Team.find({}, { apiId: 1, name: 1, area: 1, location: 1 }).lean(),
      visits(),
      trailInfo(),
    ]);
    const ov = new Set(areas.filter((a) => a.status === "blue").map((a) => a.name.toLowerCase()));
    const byArea = new Map<string, typeof markers>();
    for (const marker of markers) {
      const key = marker.area.toLowerCase();
      byArea.set(key, [...(byArea.get(key) ?? []), marker]);
    }

    const result: AreaReach[] = [];
    for (const [area, list] of [...byArea.entries()].sort(([a], [b]) => a.localeCompare(b))) {
      if (ov.has(area)) continue;
      const fix = list[list.length - 1];
      const fixCoord = fix.location.coordinates as Coord;
      const legs: SpeedLeg[] = list.slice(1).map((to, i) => ({
        id: `${String(list[i]._id)}:${String(to._id)}`, from: list[i].time, to: to.time,
        fromCoord: list[i].location.coordinates as Coord, toCoord: to.location.coordinates as Coord,
      }));
      const areaEvents = events.filter((e) => e.area.toLowerCase() === area).map((e) => ({ status: e.status, at: e.at }));
      const speed = measureSpeed(legs, trails, areaEvents, at);

      const sinceMs = Math.max(0, at.getTime() - fix.time.getTime());
      const capped = sinceMs > MAX_WALK_MS;
      const walkedM = (speed.kmh * 1000 * Math.min(sinceMs, MAX_WALK_MS)) / 3600_000;
      const distanceM = Math.max(ROUND_M, Math.round(walkedM / ROUND_M) * ROUND_M);
      const night = isNightAt(at, fixCoord[1], fixCoord[0]);
      const { polygon, estimate } = await areaShape(String(fix._id), night ? "foot_night" : "foot", fixCoord, distanceM);

      const groups: Group[] = teams
        .filter((team) => team.area?.toLowerCase() === area)
        .map((team) => ({ teamApiId: team.apiId, name: team.name, coord: team.location.coordinates as Coord }));
      const visited = new Set(visitList.filter((v) => v.area.toLowerCase() === area && v.state === "visited").map((v) => v.teamApiId));
      const target = nearestUnvisited(fixCoord, groups, visited);
      const conePolygon = target ? coneIn(fixCoord, target, polygon, distanceM + 50) : null;

      result.push({
        area,
        fix: { id: String(fix._id), type: fix.type as "hint" | "hunt", at: fix.time.toISOString(), lng: fixCoord[0], lat: fixCoord[1] },
        speedKmh: speed.kmh, measured: speed.measured, legs: speed.legs,
        distanceM, capped, night, estimate, polygon,
        cone: target && conePolygon ? { ...target, polygon: conePolygon } : null,
      });
    }
    return { enabled: true, areas: result };
  }

  async function current() {
    if (memo && now().getTime() - memo.at < MEMO_MS) return memo;
    if (inflight) return inflight;
    inflight = compute()
      .then((value) => (memo = { at: now().getTime(), value, body: JSON.stringify(value) }))
      .finally(() => { inflight = null; });
    return inflight;
  }

  return {
    async reach(): Promise<ReachResponse> { return (await current()).value; },
    async reachJson(): Promise<string> { return (await current()).body; },
  };
}
```

```ts
import type { Feature, FeatureCollection, MultiPolygon, Polygon } from 'geojson';

export type AreaGeometry = Polygon | MultiPolygon;

/** One fox's reachable area since its last spot (GET /routes/reach). */
export interface AreaReach {
  area: string;
  fix: { id: string; type: 'hint' | 'hunt'; at: string; lng: number; lat: number };
  speedKmh: number;
  measured: boolean;
  legs: number;
  distanceM: number;
  capped: boolean;
  night: boolean;
  estimate: boolean;
  polygon: AreaGeometry;
  cone: { teamApiId: number; name: string; bearing: number; distanceM: number; polygon: AreaGeometry } | null;
}

export interface ReachResponse { enabled: boolean; areas: AreaReach[] }

export interface ReachShapeProps { area: string; role: 'area' | 'cone'; color: string }

export const REACH_SOURCE = 'fox-reach';
export const REACH_FILL_LAYER = 'fox-reach-fill';
export const REACH_CONE_LAYER = 'fox-reach-cone';
export const REACH_LINE_LAYER = 'fox-reach-line';

/** "52 min", from an hour "1u05m". */
export function formatElapsed(ms: number): string {
  const minutes = Math.max(0, Math.floor(ms / 60_000));
  if (minutes < 60) return `${minutes} min`;
  return `${Math.floor(minutes / 60)}u${String(minutes % 60).padStart(2, '0')}m`;
}

/** "Alpha · 52 min · 4,6 km/u"; "~5 km/u" when the speed is a default, "3u+" once the area stopped growing. */
export function reachChipText(reach: AreaReach, now: number): string {
  const name = reach.area.charAt(0).toUpperCase() + reach.area.slice(1);
  const elapsed = reach.capped ? '3u+' : formatElapsed(now - Date.parse(reach.fix.at));
  const speed = `${reach.measured ? '' : '~'}${reach.speedKmh.toLocaleString('nl-NL', { maximumFractionDigits: 1 })} km/u`;
  return `${name} · ${elapsed} · ${speed}`;
}

/** The areas and cones as one collection; the layers filter on `role`. */
export function reachFeatures(areas: AreaReach[], colorFor: (area: string) => string): FeatureCollection<AreaGeometry, ReachShapeProps> {
  const features: Feature<AreaGeometry, ReachShapeProps>[] = [];
  for (const reach of areas) {
    const color = colorFor(reach.area);
    features.push({ type: 'Feature', geometry: reach.polygon, properties: { area: reach.area, role: 'area', color } });
    if (reach.cone) features.push({ type: 'Feature', geometry: reach.cone.polygon, properties: { area: reach.area, role: 'cone', color } });
  }
  return { type: 'FeatureCollection', features };
}
```

```tsx
import { useEffect, useMemo, useState } from 'react';
import { AnimatePresence, motion } from 'motion/react';
import { Layer, Marker, Source, type FillLayerSpecification, type LineLayerSpecification } from 'react-map-gl/maplibre';
import { useAreas } from '@/hooks/areas.hook';
import { useHuntableAreas } from '@/hooks/huntable-areas.hook';
import { useNow } from '@/hooks/now.hook';
import { useReach } from '@/hooks/reach.hook';
import { REACH_CONE_LAYER, REACH_FILL_LAYER, REACH_LINE_LAYER, REACH_SOURCE, reachChipText, reachFeatures, type AreaReach } from '@/lib/reach';
import { getColorFromArea } from '@/lib/utils';

const AREA_FILTER: FillLayerSpecification['filter'] = ['==', ['get', 'role'], 'area'];
const CONE_FILTER: FillLayerSpecification['filter'] = ['==', ['get', 'role'], 'cone'];
const LINE_FILTER: LineLayerSpecification['filter'] = ['==', ['get', 'role'], 'area'];
const FADE_MS = 250;
const FADE = { duration: FADE_MS, delay: 0 };

/**
 * Where each fox can have walked since its last hint/hunt (look A): its area lightly filled with a dashed edge, a darker
 * cone toward the nearest unvisited group, and a chip with the time since and the fox's speed. Stays mounted so turning
 * the layer off fades it out.
 */
export default function Reach({ visible }: { visible: boolean }) {
  const { areas } = useReach(visible);
  const { isVisible } = useAreas();
  const { isForbidden } = useHuntableAreas();
  const now = useNow(visible, 30_000);

  // Keep the last areas while fading out, then drop them
  const [kept, setKept] = useState<AreaReach[]>([]);
  if (visible && areas && areas !== kept) setKept(areas);
  useEffect(() => {
    if (visible) return;
    const timer = setTimeout(() => setKept([]), FADE_MS);
    return () => clearTimeout(timer);
  }, [visible]);

  const shown = useMemo(() => kept.filter((reach) => isVisible(reach.area) && !isForbidden(reach.area)), [kept, isVisible, isForbidden]);
  const data = useMemo(() => reachFeatures(shown, getColorFromArea), [shown]);

  return (
    <>
      <Source id={REACH_SOURCE} type="geojson" data={data}>
        <Layer id={REACH_FILL_LAYER} type="fill" filter={AREA_FILTER}
               paint={{ 'fill-color': ['get', 'color'], 'fill-opacity': visible ? 0.12 : 0, 'fill-opacity-transition': FADE }} />
        <Layer id={REACH_CONE_LAYER} type="fill" filter={CONE_FILTER}
               paint={{ 'fill-color': ['get', 'color'], 'fill-opacity': visible ? 0.22 : 0, 'fill-opacity-transition': FADE }} />
        <Layer id={REACH_LINE_LAYER} type="line" filter={LINE_FILTER}
               paint={{ 'line-color': ['get', 'color'], 'line-width': 2, 'line-dasharray': [3, 2], 'line-opacity': visible ? 1 : 0, 'line-opacity-transition': FADE }} />
      </Source>
      <AnimatePresence>
        {visible && shown.map((reach) => (
          <Marker key={reach.area} longitude={reach.fix.lng} latitude={reach.fix.lat} anchor="left" offset={[-7, 0]} style={{ pointerEvents: 'none' }}>
            <motion.div
              initial={{ opacity: 0, scale: 0.8 }}
              animate={{ opacity: 1, scale: 1 }}
              exit={{ opacity: 0, scale: 0.8 }}
              transition={{ duration: 0.2 }}
              className="flex origin-left items-center gap-1.5"
            >
              <span className="size-3.5 rounded-full shadow-md ring-[2.5px] ring-card" style={{ backgroundColor: getColorFromArea(reach.area) }} />
              <span className="rounded-full bg-card px-2.5 py-1 text-xs font-semibold whitespace-nowrap text-foreground tabular-nums shadow-md">
                {reachChipText(reach, now)}
              </span>
            </motion.div>
          </Marker>
        ))}
      </AnimatePresence>
    </>
  );
}
```
